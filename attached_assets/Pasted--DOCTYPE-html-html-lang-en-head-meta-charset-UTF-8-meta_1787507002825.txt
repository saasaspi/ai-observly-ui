<script>
  const tooltip = document.getElementById('tooltip');
  const card = document.getElementById('card');
  const tName = document.getElementById('t-name');
  const tCat = document.getElementById('t-cat');
  const tDesc = document.getElementById('t-desc');
  const catColors = {own:'#2563eb', direct:'#0ea5b7', obs:'#64748b', finops:'#6366f1'};

  document.querySelectorAll('.point').forEach(pt => {
    pt.addEventListener('mouseenter', (e) => { showTip(pt, e); });
    pt.addEventListener('mousemove', positionTooltip);
    pt.addEventListener('mouseleave', () => { tooltip.style.display = 'none'; });
    pt.addEventListener('click', (e) => {
      if (tooltip.style.display === 'block' && tName.textContent === pt.dataset.name) {
        tooltip.style.display = 'none';
      } else {
        showTip(pt, e);
      }
    });
  });

  function showTip(pt, e){
    const d = pt.dataset;
    tName.textContent = d.name;
    tCat.textContent = d.catlabel;
    tCat.style.color = catColors[d.cat];
    tDesc.textContent = d.desc;
    tooltip.style.display = 'block';
    positionTooltip(e);
  }

  function positionTooltip(e){
    const rect = card.getBoundingClientRect();
    let x = e.clientX - rect.left + 16;
    let y = e.clientY - rect.top + 16;
    if (x + 260 > rect.width) x = e.clientX - rect.left - 266;
    tooltip.style.left = x + 'px';
    tooltip.style.top = y + 'px';
  }

  document.querySelectorAll('.chip').forEach(chip => {
    chip.addEventListener('click', () => {
      chip.classList.toggle('off');
      const active = [...document.querySelectorAll('.chip:not(.off)')].map(c => c.dataset.filter);
      document.querySelectorAll('.point').forEach(pt => {
        pt.classList.toggle('dimmed', active.length > 0 && !active.includes(pt.dataset.cat));
      });
    });
  });
</script>
